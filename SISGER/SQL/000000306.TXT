DECLARE
  PROCEDURE AddCol(pCol VARCHAR2, pTipo VARCHAR2) IS
  BEGIN
    EXECUTE IMMEDIATE 'ALTER TABLE NFEDUP ADD (' || pCol || ' ' || pTipo || ')';
  EXCEPTION
    WHEN OTHERS THEN
      IF SQLCODE != -1430 THEN -- Ignora ORA-01430 (coluna já existe)
        RAISE;
      END IF;
  END;
BEGIN
  AddCol('CHV_NFE',     'VARCHAR2(44)');
  AddCol('NFAT',        'VARCHAR2(60)');
  AddCol('VORIG',       'NUMBER(13,2) DEFAULT 0');
  AddCol('VDESC',       'NUMBER(13,2) DEFAULT 0');
  AddCol('VLIQ',        'NUMBER(13,2) DEFAULT 0');
  AddCol('TIPO',        'VARCHAR2(1) DEFAULT ''R''');
  AddCol('COD_CLI_FOR', 'NUMBER(7,0) DEFAULT 0');
  AddCol('NR_DOC',      'NUMBER(9,0) DEFAULT 0');
  AddCol('DT_EMISSAO',  'DATE DEFAULT SYSDATE');
  AddCol('STATUS_FIN',  'VARCHAR2(1) DEFAULT ''A''');
  AddCol('DT_BAIXA',    'DATE');
  AddCol('VPAGO',       'NUMBER(15,2) DEFAULT 0');
  AddCol('VJUROS',      'NUMBER(13,2) DEFAULT 0');
  AddCol('VDESC_BAIXA', 'NUMBER(13,2) DEFAULT 0');
  AddCol('ORIGEM',      'VARCHAR2(1) DEFAULT ''N''');
  AddCol('HISTORICO',   'VARCHAR2(100)');
END;

$
