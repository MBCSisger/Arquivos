CREATE TABLE PROD_FOR (
  COD_PROD       NUMBER(6,0) NOT NULL,
  COD_FORNECEDOR VARCHAR2(40) NOT NULL,
  COD_FORNECE    NUMBER NOT NULL,
  CONSTRAINT PK_PROD_FOR PRIMARY KEY (COD_PROD, COD_FORNECEDOR, COD_FORNECE)
)
$

COMMENT ON TABLE PROD_FOR IS 'Mapeamento B2B SiSGER: Produto Interno x Codigo XML x ID Fornece'
$

CREATE INDEX IDX_PRODFOR_FORNECE ON PROD_FOR (COD_FORNECEDOR, COD_FORNECE)
$

CREATE INDEX IDX_PRODFOR_PROD ON PROD_FOR (COD_PROD)
$
